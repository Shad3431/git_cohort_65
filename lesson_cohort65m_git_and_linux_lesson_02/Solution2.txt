1. Find all students who have at least one quiz (type: "quiz") with a score greater than 10.

db.students.find({
	"scores": {
		$elemMatch:{
			type:'quiz',
			score:{$gt:10}
		}
	}
})

2. Retrieve only the names of students (name) who have at least one exam (type: "exam")
 with a score less than 5.

db.students.find({
	"scores": {
		$elemMatch:{
			type:'exam',
			score:{$lt:5}
		}
	}
}, {_id:0, name:1})

3. Find students who have at least one homework score (type: "homework") greater than 30.

db.students.find({
	"scores": {
		$elemMatch:{
			type:'homework',
			score:{$gt:30}
		}
	}
})

4. Retrieve only the names (name) and quiz scores (type: "quiz") for all students.

db.students.find({},{
	_id:0,
	name:1,
	"scores": {
		$elemMatch:{
			type:'quiz',
		}
	}
})

5. Find all students who have scores for exams (type: "exam") or quizzes (type: "quiz")
 and return their names and scores.

db.students.find({
	"scores": {
		$elemMatch:{
			type: {
				$in: ['exam','quiz']
			}
		}
	}
}, {_id:0, name:1, scores:1})
